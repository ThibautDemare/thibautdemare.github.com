# robots.txt du domaine thibautdemare.github.io
#
# Ce fichier vaut pour TOUT l'hôte, donc pour ce site personnel ET pour
# /Ludaskia/, qui est un autre dépôt publié sous le même domaine. Un robots.txt
# n'est lu qu'à la racine d'un hôte : celui qu'on placerait dans le dépôt
# Ludaskia, à /Ludaskia/robots.txt, ne serait lu par personne. D'où sa place ici.

# ─────────────────────────────────────────────────────────────────────────────
# Règle générale : tout est explorable, sauf les présentations.
# /presentations/ = supports de conférences en reveal.js (2015-2019). Découpés
# en fragments de diapositives, ils ressortent mal en résultat de recherche et
# n'apportent rien à qui cherche ces travaux — les publications, elles, sont
# indexées.
# À savoir : ceci empêche l'EXPLORATION, pas l'indexation. Une page déjà connue
# d'un moteur peut rester affichée sans extrait, et lui interdire l'accès
# l'empêche même de découvrir un éventuel `noindex`. Pour un vrai retrait, il
# faudrait au contraire laisser explorer et poser un `noindex`.
User-agent: *
Allow: /
Disallow: /presentations/

# ─────────────────────────────────────────────────────────────────────────────
# Moteurs de recherche et assistants de RECHERCHE : bienvenus.
# Ils sont listés à part pour rendre l'intention lisible, mais leurs règles sont
# les MÊMES que ci-dessus — et c'est un piège à connaître : un robot qui trouve
# un groupe à son nom applique CE groupe et ignore complètement `User-agent: *`.
# Oublier d'y répéter `Disallow: /presentations/` laisserait donc ces robots-là
# explorer les présentations.
User-agent: Googlebot
User-agent: Bingbot
User-agent: DuckDuckBot
User-agent: Qwantify
User-agent: OAI-SearchBot
User-agent: Claude-SearchBot
User-agent: PerplexityBot
Allow: /
Disallow: /presentations/

# ─────────────────────────────────────────────────────────────────────────────
# Collecte destinée à l'ENTRAÎNEMENT de modèles : refusée.
# La distinction est volontaire : être trouvable et cité par un assistant est un
# canal de découverte utile (les robots ci-dessus), alimenter l'entraînement d'un
# modèle en est un autre, sans contrepartie. Ludaskia est par ailleurs sous
# licence propriétaire, tous droits réservés.
# Ces règles reposent sur la bonne volonté des robots : elles n'empêchent
# techniquement rien, elles déclarent une intention que les acteurs sérieux
# respectent.
User-agent: GPTBot
Disallow: /

User-agent: ClaudeBot
Disallow: /

User-agent: CCBot
Disallow: /

User-agent: Bytespider
Disallow: /

User-agent: Meta-ExternalAgent

# Google-Extended et Applebot-Extended ne sont pas des explorateurs : ce sont des
# interrupteurs d'USAGE pour Gemini et Apple Intelligence, appliqués à ce que
# Googlebot et Applebot ont déjà collecté. Les refuser ne retire donc rien de la
# recherche Google ni de Siri, et ne change rien aux AI Overviews, qui s'appuient
# sur Googlebot.
